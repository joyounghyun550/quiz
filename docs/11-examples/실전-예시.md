### API 레이어

```tsx
// features/feature-name/api/feature-name.api.ts
import { axiosInstance } from "@/libs/axios";
import type { Item, CreateItemRequest, UpdateItemRequest } from "./feature-name.api.type";

export const itemApi = {
  getList: async (): Promise<Item[]> => {
    const { data } = await axiosInstance.get("/items");
    return data;
  },

  getById: async (id: string): Promise<Item> => {
    const { data } = await axiosInstance.get(`/items/${id}`);
    return data;
  },

  create: async (payload: CreateItemRequest): Promise<Item> => {
    const { data } = await axiosInstance.post("/items", payload);
    return data;
  },

  update: async (id: string, payload: UpdateItemRequest): Promise<Item> => {
    const { data } = await axiosInstance.put(`/items/${id}`, payload);
    return data;
  },

  delete: async (id: string): Promise<void> => {
    await axiosInstance.delete(`/items/${id}`);
  },
};
```

## 3. 무한 스크롤 패턴 (Infinite Scroll)

### TanStack Query의 useInfiniteQuery 활용

```tsx
// features/feature-name/hooks/query/use-infinite-items-query.ts
import { useInfiniteQuery } from "@tanstack/react-query";
import { itemApi } from "@/features/feature-name/api";

export function useInfiniteItemsQuery() {
  return useInfiniteQuery({
    queryKey: ["items", "infinite"],
    queryFn: ({ pageParam = 1 }) => itemApi.getList({ page: pageParam }),
    getNextPageParam: (lastPage, allPages) => {
      return lastPage.hasMore ? allPages.length + 1 : undefined;
    },
    initialPageParam: 1,
  });
}
```

### 무한 스크롤 컴포넌트

```tsx
// features/feature-name/components/InfiniteItemList.tsx
"use client";

import { useInfiniteItemsQuery } from "@/features/feature-name/hooks";
import { useInfiniteScroll } from "@/shared/hooks/use-infinite-scroll";

export function InfiniteItemList() {
  const { data, fetchNextPage, hasNextPage, isFetchingNextPage } = useInfiniteItemsQuery();

  const loadMoreRef = useInfiniteScroll({
    onIntersect: fetchNextPage,
    enabled: hasNextPage && !isFetchingNextPage,
  });

  return (
    <div>
      {data?.pages.map((page) => page.items.map((item) => <ItemCard key={item.id} item={item} />))}
      <div ref={loadMoreRef}>{isFetchingNextPage && <Spinner />}</div>
    </div>
  );
}
```

## 4. 폼 처리 패턴 (Form Handling)

### 폼 컴포넌트 예시

```tsx
// features/feature-name/components/ItemForm.tsx
"use client";

import { useState } from "react";
import { useCreateItemMutation } from "@/features/feature-name/hooks";
import { toast } from "sonner";

export function ItemForm() {
  const [title, setTitle] = useState("");
  const [description, setDescription] = useState("");

  const createMutation = useCreateItemMutation();

  const handleSubmit = async (e: React.FormEvent) => {
    e.preventDefault();

    try {
      await createMutation.mutateAsync({ title, description });
      toast.success("생성되었습니다.");
      setTitle("");
      setDescription("");
    } catch (error) {
      toast.error("생성에 실패했습니다.");
    }
  };

  return (
    <form onSubmit={handleSubmit} className="flex flex-col gap-4">
      <input type="text" value={title} onChange={(e) => setTitle(e.target.value)} placeholder="제목" required />
      <textarea value={description} onChange={(e) => setDescription(e.target.value)} placeholder="설명" required />
      <button type="submit" disabled={createMutation.isPending}>
        {createMutation.isPending ? "생성 중..." : "생성"}
      </button>
    </form>
  );
}
```

## 5. Server + Client Hybrid 패턴

서버에서 초기 데이터를 페칭하고, 클라이언트에서 인터랙션을 처리하는 패턴:

```tsx
// app/items/page.tsx - Server Component
import { ItemList } from "@/features/feature-name/components";
import { SearchBar } from "@/features/feature-name/components/SearchBar"; // Client Component

/**
 * 아이템 목록 페이지
 * 서버에서 초기 데이터를 페칭하고, 클라이언트에서 인터랙션 처리합니다.
 */
export default async function ItemsPage() {
  // ✅ 서버에서 초기 데이터 페칭
  const initialItems = await fetch("https://api.example.com/items").then((res) => res.json());

  return (
    <div>
      <h1>아이템 목록</h1>

      {/* ✅ 검색 인터랙션 - Client Component */}
      <SearchBar />

      {/* ✅ 목록 표시 - Server Component 데이터를 Client Component에 전달 */}
      <ItemList initialData={initialItems} />
    </div>
  );
}
```

## 6. 동적 라우트 + 메타데이터 패턴

```tsx
// app/items/[id]/page.tsx
import type { Metadata } from "next";
import { itemApi } from "@/features/feature-name/api";

type Props = {
  params: { id: string };
  searchParams: { [key: string]: string | string[] | undefined };
};

export async function generateMetadata({ params }: Props): Promise<Metadata> {
  const item = await itemApi.getById(params.id);

  return {
    title: item.title,
    description: item.description,
  };
}

export default function ItemDetailPage({ params, searchParams }: Props) {
  return <div>아이템 ID: {params.id}</div>;
}
```

## 📌 참고: 실제 프로젝트 구현 사례 (buildin-web-front)

이 boilerplate를 사용한 실제 프로젝트에서의 구현 예시:

### 1. 게시글 기능 (features/post/)

- **목록 페이지**: `app/posts/page.tsx` - 게시글 목록 + 페이지네이션
- **상세 페이지**: `app/posts/[id]/page.tsx` - 게시글 상세 내용
- **API**: `post.api.ts` - 게시글 CRUD API
- **훅**: `use-post-list-query.ts`, `use-post-detail-query.ts`

### 2. 채팅 기능 (features/chat/)

- **페이지**: `app/chat/page.tsx` - 실시간 채팅 인터페이스
- **세션 관리**: `use-delete-chat-session-mutation.ts` - 채팅 세션 삭제
- **유틸리티**: `message-formatter.util.ts` - 메시지 포맷팅, `chat-history.util.ts` - 채팅 히스토리 관리

### 3. 인증 기능 (features/auth/)

- **로그인**: `app/(auth)/login/page.tsx` - 소셜 로그인
- **약관 동의**: `TermsAgreementDialog.tsx` - 초기 약관 동의 다이얼로그
- **회원 탈퇴**: `use-withdraw-mutation.ts` - 회원 탈퇴 플로우
- **OAuth 콜백**: `app/(auth)/auth/callback/page.tsx` - OAuth 리다이렉트 처리

### 4. 사이드바 기능 (features/sidebar/)

- **조건부 레이아웃**: `use-conditional-layout.ts` - 경로에 따른 사이드바 표시 여부
- **설정 관리**: `use-chat-setting.ts` - 채팅 설정 토글
