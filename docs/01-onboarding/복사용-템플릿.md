# 복사용 템플릿

📍 **위치**: [홈](../00-README.md) > [온보딩](../01-onboarding/) > 복사용 템플릿

> 💡 **Tip**: 아래 템플릿을 복사해서 `[ComponentName]`, `[Entity]` 등을 실제 이름으로 바꾸세요

---

## 새 컴포넌트 생성 시

### 컴포넌트 파일

```tsx
// features/[도메인]/components/[ComponentName].tsx
"use client";

import type { [ComponentName]Props } from "./[ComponentName].type";

/**
 * [컴포넌트 설명을 여기에 작성]
 *
 * @param props - [ComponentName]Props
 * @example
 * <[ComponentName] />
 */
const [ComponentName] = ({}: [ComponentName]Props) => {
  return (
    <div>
      {/* 내용 */}
    </div>
  );
};

export default [ComponentName];
```

### 타입 파일

```tsx
// features/[도메인]/components/[ComponentName].type.ts
/**
 * [ComponentName] 컴포넌트의 Props
 */
export type [ComponentName]Props = {
  /** 필드 설명 */
  fieldName: string;
};
```

## 새 커스텀 훅 생성 시

```tsx
// features/[도메인]/hooks/use-[hook-name].ts
"use client";

import { useState, useCallback } from "react";

/**
 * [훅 설명을 여기에 작성]
 *
 * @param param - 파라미터 설명
 * @returns 반환값 설명
 *
 * @example
 * const { data, handler } = use[HookName]();
 */
export const use[HookName] = (param: ParamType) => {
  const [state, setState] = useState<StateType>(initialValue);

  const handler = useCallback(() => {
    // 로직
  }, []);

  return {
    state,
    handler,
  };
};
```

## 새 Query 훅 생성 시

```tsx
// features/[도메인]/hooks/query/use-[entity]-query.ts
import { useQuery } from "@tanstack/react-query";
import { get[Entity] } from "@/features/[도메인]/api";

/**
 * [Entity] 데이터를 조회하는 Query 훅
 *
 * @param id - 조회할 [Entity]의 ID
 * @returns Query 결과 (data, isLoading, error 등)
 *
 * @example
 * const { data, isLoading } = use[Entity]Query(id);
 */
export const use[Entity]Query = (id: string) => {
  return useQuery({
    queryKey: ["[entity]", id],
    queryFn: () => get[Entity](id),
    enabled: !!id,
  });
};
```

## 새 Mutation 훅 생성 시

```tsx
// features/[도메인]/hooks/mutation/use-create-[entity]-mutation.ts
import { useMutation, useQueryClient } from "@tanstack/react-query";
import { create[Entity] } from "@/features/[도메인]/api";

/**
 * [Entity] 생성 Mutation 훅
 *
 * @returns Mutation 결과 (mutate, isLoading, error 등)
 *
 * @example
 * const { mutate } = useCreate[Entity]Mutation();
 * mutate({ name: "새 항목" });
 */
export const useCreate[Entity]Mutation = () => {
  const queryClient = useQueryClient();

  return useMutation({
    mutationFn: create[Entity],
    onSuccess: () => {
      queryClient.invalidateQueries({ queryKey: ["[entity]"] });
    },
  });
};
```

## 사용 예시

### 실제 컴포넌트 예시

```tsx
// features/post/components/PostCard.tsx
"use client";

import { SolidPrimary } from "@itandsy/react-common";
import type { PostCardProps } from "./PostCard.type";

/**
 * 게시글 카드 컴포넌트
 *
 * @param props - PostCardProps
 * @example
 * <PostCard title="제목" content="내용" />
 */
const PostCard = ({ title, content }: PostCardProps) => {
  const handleClick = () => {
    console.log("클릭됨");
  };

  return (
    <div className="rounded-lg border p-4">
      <h3 className="text-lg font-bold">{title}</h3>
      <p className="mt-2 text-gray-600">{content}</p>
      <SolidPrimary label="자세히 보기" onClick={handleClick} />
    </div>
  );
};

export default PostCard;
```

### 실제 Query 훅 예시

```tsx
// features/post/hooks/query/use-post-query.ts
import { useQuery } from "@tanstack/react-query";
import { getPost } from "@/features/post/api";

/**
 * 게시글 데이터를 조회하는 Query 훅
 *
 * @param id - 조회할 게시글 ID
 * @returns Query 결과 (data, isLoading, error 등)
 *
 * @example
 * const { data, isLoading } = usePostQuery("123");
 */
export const usePostQuery = (id: string) => {
  return useQuery({
    queryKey: ["post", id],
    queryFn: () => getPost(id),
    enabled: !!id,
  });
};
```
